Question: Which team scored the least goals in a single season across all your database and how many?
SQL:
SELECT Season, Team, SUM(Goals) AS TotalGoals
FROM (
    SELECT Season, HomeTeam AS Team, FullTimeHomeGoals AS Goals FROM premier_league
    UNION ALL
    SELECT Season, AwayTeam AS Team, FullTimeAwayGoals AS Goals FROM premier_league
)
GROUP BY Season, Team
ORDER BY TotalGoals ASC LIMIT 1;

Question: What was the best team in the 2012/13 season?
SQL: 
SELECT Team, SUM(Points) AS TotalPoints 
FROM (
    SELECT HomeTeam AS Team, 
           CASE WHEN FullTimeResult = 'H' THEN 3 WHEN FullTimeResult = 'D' THEN 1 ELSE 0 END AS Points 
    FROM premier_league WHERE Season = '2012/13'
    UNION ALL
    SELECT AwayTeam AS Team, 
           CASE WHEN FullTimeResult = 'A' THEN 3 WHEN FullTimeResult = 'D' THEN 1 ELSE 0 END AS Points 
    FROM premier_league WHERE Season = '2012/13'
) 
GROUP BY Team 
ORDER BY TotalPoints DESC LIMIT 1;
===
Question: Which team recorded the least goals conceded at home and away throughout all seasons?
SQL: 
SELECT Team, SUM(GoalsConceded) AS TotalConceded 
FROM (
    SELECT HomeTeam AS Team, FullTimeAwayGoals AS GoalsConceded FROM premier_league
    UNION ALL
    SELECT AwayTeam AS Team, FullTimeHomeGoals AS GoalsConceded FROM premier_league
) 
GROUP BY Team 
HAVING COUNT(*) > 38 -- Filters out teams that only played one season
ORDER BY TotalConceded ASC LIMIT 1;
